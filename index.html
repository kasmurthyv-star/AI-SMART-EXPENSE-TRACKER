Production Implementation Architecture

3 ways to build an automatic PhonePe / Paytm expense tracker in India

1. NotificationListener (Android)

Captures push alerts from `com.phonepe.app`, `net.one97.paytm`, and bank SMS in real time without Play Store SMS restriction issues.

2. UPI Intent Wrapper

Invokes PhonePe/Paytm via `upi://pay` URI, reads the `ActivityResult` (`Status=SUCCESS&txnId=...`), and saves the category immediately.

3. Account Aggregator (Setu/Finvu)

Works on both iOS & Android. Fetches consent-based bank statements directly from HDFC, SBI, ICICI, and Axis FIPs.

// Android NotificationListenerService for PhonePe, Paytm & Bank UPI Alerts
class UpiExpenseNotificationService : NotificationListenerService() {
    private val upiPackages = setOf(
        "com.phonepe.app", "net.one97.paytm",
        "com.google.android.apps.nbu.paisa.user"
    )
    private val amtRegex = Regex("""(?:Rs\.?|INR|₹)\s*([0-9,]+(?:\.\d{1,2})?)""", RegexOption.IGNORE_CASE)
    private val merchantRegex = Regex("""(?:to|at|paid to|trf to)\s+([A-Za-z0-9 &._-]+?)(?:\s+on|\s+via|\s+Ref|\s+UPI|\.)""", RegexOption.IGNORE_CASE)

    override fun onNotificationPosted(sbn: StatusBarNotification) {
        val extras = sbn.notification.extras
        val text = extras.getCharSequence("android.text")?.toString() ?: return
        if (!text.contains("debited", true) && !text.contains("paid", true)) return

        val amount = amtRegex.find(text)?.groupValues?.get(1)?.replace(",", "")?.toDoubleOrNull() ?: return
        val merchant = merchantRegex.find(text)?.groupValues?.get(1)?.trim() ?: "UPI Merchant"
        val sourceApp = when (sbn.packageName) {
            "com.phonepe.app" -> "PhonePe"
            "net.one97.paytm" -> "Paytm"
            else -> "Bank SMS / GPay"
        }
        ExpenseRepository.autoCategorizeAndSave(amount, merchant, sourceApp, text)
    }
}
